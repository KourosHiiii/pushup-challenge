    44→/** سلام زمان‌محور بر اساس ساعت تهران — محاسبه در اولین رندر (تهران-محور روی سرور و کلاینت یکسانه) */
    45→function computeGreeting(): string {
    46→  try {
    47→    const hour = Number(
    48→      new Intl.DateTimeFormat("en-GB", {
    49→        timeZone: "Asia/Tehran",
    50→        hour: "numeric",
    51→        hour12: false,
    52→      }).format(new Date())
    53→    );
    54→    if (hour < 5) return "شب بخیر قهرمان!";
    55→    if (hour < 12) return "صبح بخیر قهرمان!";
    56→    if (hour < 17) return "وقت بخیر قهرمان!";
    57→    if (hour < 21) return "عصر بخیر قهرمان!";
    58→    return "شب بخیر قهرمان!";
    59→  } catch {
    60→    return "سلام قهرمان!";
    61→  }
    62→}
    63→
    64→function useGreeting() {
    65→  return useState(computeGreeting)[0];
    66→}

    72→  return "neutral";
    73→}

   485→/** دیالوگ ثبت تمرین آزاد — الگوی دیالوگ چک‌این با بازه ۱ تا ۱۰۰۰ */
   486→function FreeModeCheckinDialog({
   487→  open,
   488→  onOpenChange,
   489→  lastDone,
   490→  onSubmit,
   491→}: {
   492→  open: boolean;
   493→  onOpenChange: (v: boolean) => void;
   494→  lastDone: number;
   495→  onSubmit: (completed: number) => void;
   496→}) {
   497→  const [count, setCount] = useState(lastDone);
   498→  const [loading, setLoading] = useState(false);
   499→
   500→  useEffect(() => {
   501→    if (open) {
   502→      setCount(Math.min(1000, Math.max(1, lastDone)));
   503→      setLoading(false);
   504→    }
   505→  }, [open, lastDone]);
   506→
   507→  const clamp = (v: number) => Math.min(1000, Math.max(1, Math.round(v)));
   508→
   509→  const submit = async () => {
   510→    setLoading(true);
   511→    try {
   512→      onSubmit(clamp(count));
   513→      onOpenChange(false);
   514→    } finally {
   515→      setLoading(false);
   516→    }
   517→  };
   518→
   519→  return (
   520→    <Dialog open={open} onOpenChange={onOpenChange}>
   521→      <DialogContent className="max-w-[340px] rounded-[28px] p-5" dir="rtl">
   522→        <DialogHeader className="items-center text-center">
   523→          <DialogTitle className="text-base">ثبت تمرین آزاد</DialogTitle>
   524→          <DialogDescription className="text-[11.5px] leading-relaxed">
   525→            چند تا شنا زدی؟ هر عددی ثبت کنی استریک امروزت قفل می‌شه!
   526→          </DialogDescription>
   527→        </DialogHeader>
   528→
   529→        {/* شمارنده */}
   530→        <div className="mt-1 flex items-center justify-center gap-4">
   531→          <button
   532→            onClick={() => setCount((c) => clamp(c - 1))}
   533→            className="flex size-11 items-center justify-center rounded-2xl border border-border bg-muted/50 transition-colors hover:bg-muted active:scale-95"
   534→            aria-label="یکی کمتر"
   535→          >
   536→            <Minus className="size-4" />
   537→          </button>
   538→          <div className="flex min-w-[110px] flex-col items-center rounded-2xl bg-orange-50 px-4 py-2.5 dark:bg-orange-950/50">
   539→            <span className="text-4xl font-black tabular-nums leading-none text-orange-600 dark:text-orange-400">
   540→              {toFa(count)}
   541→            </span>
   542→            <span className="mt-1 text-[9.5px] font-bold text-muted-foreground">شنا سوئدی</span>
   543→          </div>
   544→          <button
   545→            onClick={() => setCount((c) => clamp(c + 1))}
   546→            className="flex size-11 items-center justify-center rounded-2xl border border-border bg-muted/50 transition-colors hover:bg-muted active:scale-95"
   547→            aria-label="یکی بیشتر"
   548→          >
   549→            <Plus className="size-4" />
   550→          </button>
   551→        </div>
   552→
   553→        {/* میانبرهای سریع */}
   554→        <div className="mt-2 flex items-center justify-center gap-2">
   555→          {[-5, -1, 1, 5, 10].map((v) => (
   556→            <button
   557→              key={v}
   558→              onClick={() => setCount((c) => clamp(c + v))}
   559→              className="rounded-full border border-orange-200 bg-orange-50 px-2.5 py-1.5 text-[11px] font-extrabold text-orange-600 transition-colors hover:bg-orange-100 dark:border-orange-800 dark:bg-orange-950/60 dark:text-orange-400"
   560→            >
   561→              {v > 0 ? `+${toFa(v)}` : toFa(v)}
   562→            </button>
   563→          ))}
   564→        </div>
   565→
   566→        <button
   567→          onClick={submit}
   568→          disabled={loading}
   569→          className="mt-2 flex w-full items-center justify-center gap-2 rounded-2xl bg-gradient-to-l from-orange-500 to-orange-600 py-3.5 text-[14px] font-black text-white shadow-lg shadow-orange-500/25 transition-transform active:scale-[0.98] disabled:opacity-60"
   570→        >
   571→          {loading ? <Loader2 className="size-4 animate-spin" /> : null}
   572→          ثبت تمرین آزاد
   573→        </button>
   574→      </DialogContent>
   575→    </Dialog>
   576→  );
   577→}
